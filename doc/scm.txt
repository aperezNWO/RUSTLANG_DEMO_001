# Set session paths to C:\Rust

$env:RUSTUP_HOME = "C:\Rust\rustup"
$env:CARGO_TARGET_DIR = "C:\Rust\target"
$env:Path = "C:\Rust\cargo\bin;" + $env:Path

# Remove legacy config files pointing to C:\Users\Vostro
Remove-Item -Path "C:\Rust\rustup\settings.toml" -Force -ErrorAction SilentlyContinue
Remove-Item -Path "C:\Users\Vostro\.rustup" -Recurse -Force -ErrorAction SilentlyContinue
Remove-Item -Path "C:\Users\Vostro\.cargo" -Recurse -Force -ErrorAction SilentlyContinue

# 2. Run the installer from the same session
