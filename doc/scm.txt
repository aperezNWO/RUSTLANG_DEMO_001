## powershell

## Install rustup (Rust installer)
## powershell  on "c:\Rust"
Invoke-WebRequest -Uri "https://win.rustup.rs" -OutFile "rustup-init.exe"

# 1. Define target directories for the installer process
# Set session paths to C:\Rust
$env:CARGO_HOME = "C:\Rust\cargo"
$env:RUSTUP_HOME = "C:\Rust\rustup"
$env:CARGO_TARGET_DIR = "C:\Rust\target"
$env:Path = "C:\Rust\cargo\bin;" + $env:Path

# Remove legacy config files pointing to C:\Users\Vostro
Remove-Item -Path "C:\Rust\rustup\settings.toml" -Force -ErrorAction SilentlyContinue
Remove-Item -Path "C:\Users\Vostro\.rustup" -Recurse -Force -ErrorAction SilentlyContinue
Remove-Item -Path "C:\Users\Vostro\.cargo" -Recurse -Force -ErrorAction SilentlyContinue

# 2. Run the installer from the same session

C:\Rust\rustup-init.exe -y --no-modify-path --default-host x86_64-pc-windows-gnu --default-toolchain stable-x86_64-pc-windows-gnu

## To get started you need Cargo's bin directory (C:\Rust\cargo\bin) in your PATH
## Press Win + R, type sysdm.cpl, and press Enter.

Get-Command cargo | Select-Object Source
# C:\Rust\cargo\bin\cargo.exe
